# Python dependencies
# Install with: pip install -r requirements/pip.txt
#
# --- Background removal ---
rembg
onnxruntime       # CPU inference. Swap for onnxruntime-gpu to use CUDA — no code changes needed.

# --- Video enhancement (Real-ESRGAN) ---
realesrgan
basicsr
opencv-python-headless   # cv2 for frame I/O in the enhancement module.
                         # Use opencv-python instead if you need GUI windows (e.g. cv2.imshow).

# --- PyTorch ---
# Install torch and torchvision separately using the official selector to match
# your CUDA version: https://pytorch.org/get-started/locally/
# Example (CUDA 12.1):
#   pip install torch torchvision --index-url https://download.pytorch.org/whl/cu121
# Example (CPU only):
#   pip install torch torchvision --index-url https://download.pytorch.org/whl/cpu

# --- Frame interpolation (VFIformer) ---
# VFIformer itself is cloned from GitHub (not pip-installable). The packages
# below are its Python runtime dependencies.
imageio           # Image I/O used internally by VFIformer.
scikit-image      # Image processing utilities (structural similarity, etc.).
einops            # Tensor rearrangement (rearrange/repeat) used in transformer layers.
tensorboardX      # Logging infrastructure imported by VFIformer even during inference.
# git must also be available on the system PATH for setup_vfiformer() to clone the repo.
# Download net_220.pth (~230 MB) separately:
#   https://huggingface.co/xmanifold/vfiformer/tree/main/pretrained_VFIformer

# --- Shared utilities ---
ffmpeg-python     # Python bindings for FFmpeg (frame extraction, reassembly, muxing).
tqdm              # Progress bars in terminal output.
